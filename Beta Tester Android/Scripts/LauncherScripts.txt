module Launcher
  TOKEN_A = "Token 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"
  TOKEN_B = "Token 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"

  module_function 
  def pbDownloadToFile_Launcher(url = "",filename ="", directory = "")
    headers = {
      "Proxy-Connection" => "Close",
      "Pragma" => "no-cache",
      "User-Agent" => "Mozilla/5.0 (Windows; U; Windows NT 5.1; en-US; rv:1.9.0.14) Gecko/2009082707 Firefox/3.0.14",
    }
    headers["Authorization"] = TOKEN_A 
    begin
      pbSetWindowText("Baixando Conteúdo... - #{filename}")
      response = HTTPLite.get(url, headers) 
      if response[:status] == 429
        headers["Authorization"] = TOKEN_B
        response = HTTPLite.get(url, headers)
        if response[:status] == 429
          pbIntroMessage("O servidor está extremamente congestionado, tente novamente mais tarde.")
          pbIntroMessage("O servidor está extremamente congestionado, tente novamente mais tarde.") if $adm_mode
          !exit
        end
      end
    rescue => e 
      pbIntroMessage("Houve uma falha na requisicao, verifique sua conexao com a internet ou tente mais tarde, o servidor pode estar congestionado.")
      pbIntroMessage("Houve uma falha na requisicao, verifique sua conexao com a internet ou tente mais tarde, o servidor pode estar congestionado.") if $adm_mode
      !exit
    end
    pbSetWindowText("Instalando Conteúdo de #{filename}...")
    File.open(directory + filename, "wb") { |f| f.write(response[:body]) }
    return ""
  end
end

module UpdaterDownloaderArchives
  BASE_URL = "https://raw.githubusercontent.com/messi3d/PokeFallen-UPDATES/main/"
  FONTS = ["Bitsy.ttf","Black White 2.ttf","Roma Bold.otf"]

  module_function
  def update_exec
    ret = false 
    #if File.size("Fallen.exe") != 14874112
    #  pbIntroMessage("Seu PokeFallen está desatualizado...")
    #  pbIntroMessage("Atualize para a versão oficial pelo site do projeto!")
    #  pbIntroMessage("Faça uma cópia do seu save para esta atualização!")
    #  launch_joiplay_download
    #  exit 
    #end
    if File.size("x64-msvcrt-ruby310.dll") != 10155536
      url = BASE_URL + "Beta Tester/x64-msvcrt-ruby310.dll"
      pbDownloadToFile_Launcher(url, "x64-msvcrt-ruby310.dll", "")
      ret = true 
    end 
    if File.size("zlib1.dll") != 120814
      url = BASE_URL + "Beta Tester/zlib1.dll"
      pbDownloadToFile_Launcher(url, "zlib1.dll", "")
      ret = true 
    end 
    if !File.exist?("Fallen.ini") || File.size("Fallen.ini") != 123 && File.size("Fallen.ini") != 115 
      url = BASE_URL + "Launcher PokeFallen/Fallen.ini"
      Launcher.pbDownloadToFile_Launcher(url, "Fallen.ini", "")
      ret = true 
    end
    if ret 
      pbIntroMessage("Executável atualizado com sucesso, reiniciando...")
      exit 
    end 
  end
  
  def update_patches
    return false 
    url = BASE_URL + "Beta Tester Android/patches.json"
    if !File.exist?("patches.json") || File.size("patches.json") != 460 
      Launcher.pbDownloadToFile_Launcher(url, "patches.json", "")
      pbIntroMessage("Patches atualizados com sucesso, reiniciando...")
      !exit
    end
  end
  
  def update_fonts
    fonts = FONTS
    reload = false
    fonts.each do |filename|
      path = "Fonts/#{filename}"
      next if File.exist?(path)
      url = BASE_URL + "#{path}"
      Launcher.pbDownloadToFile_Launcher(url, filename, "Fonts/")
      reload = true
    end
    return reload
  end

  # Data
  def update_data
    url = BASE_URL + "Beta Tester Android/Data/Scripts.rxdata"
    if File.size("Data/Scripts.rxdata") != 4331
      Launcher.pbDownloadToFile_Launcher(url, "Scripts.rxdata", "Data/")
      pbIntroMessage("Data Dados atualizados com sucesso, reiniciando...")
      !exit
    end
  end 
end

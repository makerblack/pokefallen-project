def pbSetResizeFactor(factor)
  if !$ResizeInitialized
    Graphics.resize_screen(Settings::SCREEN_WIDTH, Settings::SCREEN_HEIGHT)
    $ResizeInitialized = true
  end
  if factor < 0 || factor == 4
    Graphics.fullscreen = true if !Graphics.fullscreen
  else
    Graphics.fullscreen = false if Graphics.fullscreen
    Graphics.scale = (factor + 1) * 0.5
    Graphics.center
  end
end

def fallen_define_constants!(data)
  ids     = data["ids"]     || {}
  scripts = data["scripts"] || {}
  d       = data["data"]    || {}

  const_map = {
    :FALLEN_0S        => ids["FALLEN_0S"].to_i,
    :FALLEN_0G        => ids["FALLEN_0G"].to_i,
    :FALLEN_DLC       => ids["FALLEN_DLC"].to_i,
    :DATA_S_0         => d["DATA_S_0"].to_i,
    :DATA_0           => d["DATA_0"].to_i,
    :SCRIPTS_FALLEN   => scripts["SCRIPTS_FALLEN"].to_i,
    :SCRIPTS_FALLEN_A => scripts["SCRIPTS_FALLEN_A".to_i]
  }

  # Expande FALLEN_0G_1..6 a partir da lista
  if ids["FALLEN_0G_LIST"].is_a?(Array)
    ids["FALLEN_0G_LIST"].each_with_index do |val, i|
      const_map[:"FALLEN_0G_#{i + 1}"] = val.to_i
    end
  end

  const_map.each do |name, value|
    next if value.nil?
    if Object.const_defined?(name)
      current = Object.const_get(name)
      next if current == value  
      Object.send(:remove_const, name) rescue nil
    end
    Object.const_set(name, value)
  end
end

  def self.load_event(event_key, initial_time, final_time, mode)
    if FallenTime.beetween_days(initial_time, final_time)
      EVENTS[event_key][:active] = true 
      load_message(event_key) if mode == "Intro"
    else
      EVENTS[event_key][:active] = false 
    end
    if !FallenTime.beetween_days_offline(initial_time, final_time)
      EVENTS[event_key][:active] = false 
    end
  end
  
  def self.load_message(event_key)
    event = EVENTS[event_key]
    if event[:active] 
      pbIntroMessage(event[:message]) 
    end
  end

      return false

module Launcher
  TOKEN_A = "Token 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"
  TOKEN_B = "Token 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"
  TOKEN_C = "Token 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"
  TOKEN_D = "Token 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"
  
  module_function 
  def pbDownloadToFile_Launcher(url = "",filename ="", directory = "")
    headers = {
      "Proxy-Connection" => "Close",
      "Pragma" => "no-cache",
      "User-Agent" => "Mozilla/5.0 (Windows; U; Windows NT 5.1; en-US; rv:1.9.0.14) Gecko/2009082707 Firefox/3.0.14",
    }
    headers["Authorization"] = TOKEN_A 
    begin
      pbSetWindowText("Baixando Conteúdo... - #{filename}")
      response = HTTPLite.get(url, headers) 
      if response[:status] == 429
        headers["Authorization"] = TOKEN_B
        response = HTTPLite.get(url, headers)
        if response[:status] == 429
          headers["Authorization"] = TOKEN_C
          response = HTTPLite.get(url, headers) 
          if response[:status] == 429
            headers["Authorization"] = TOKEN_D
            response = HTTPLite.get(url, headers) 
            if response[:status] == 429
              pbIntroMessage("O servidor está extremamente congestionado, tente novamente mais tarde.")
              p "O servidor está extremamente congestionado, tente novamente mais tarde." if $adm_mode
              !exit
            end
          end
        end
      end
    rescue => e 
      pbIntroMessage("Houve uma falha na requisicao, verifique sua conexao com a internet ou tente mais tarde, o servidor pode estar congestionado.")
      p "Houve uma falha na requisicao, verifique sua conexao com a internet ou tente mais tarde, o servidor pode estar congestionado." if $adm_mode
      !exit
    end
    pbSetWindowText("Instalando Conteúdo de #{filename}...")
    File.open(directory + filename, "wb") { |f| f.write(response[:body]) }
    return ""
  end
end

    @bubble2.opacity=0
  end

  def pbFallenMessageinChoice(message,waiting=50)
    @overlay2.clear if @overlay2
    drawTextEx(@sprites["overlay2"].bitmap,
    @bubble2.x+27,@bubble2.y+14,465,3,message,@baseColor,@shadowColor)
    10.times do 
      pbUpdate
      Graphics.update
      @bubble2.opacity+=25.5
      @bubble2.y-=2
      @sprites["overlay2"].opacity+=25.5
      @sprites["overlay2"].y-=2
    end
    time=0
    loop do
      pbUpdate
      Graphics.update
      Input.update
      time+=1
      break if time>waiting || Input.trigger?(Input::C) || Input.trigger?(Input::BACK) 
    end
    10.times do 
      pbUpdate
      Graphics.update
      @bubble2.opacity-=25.5
      @bubble2.y+=2
      @sprites["overlay2"].opacity-=25.5
      @sprites["overlay2"].y+=2
    end
  end

  def wait(numFrames)
    numFrames.times do
      Graphics.update
      Input.update
    end
  end

  def pbUpdate
    pbUpdateSpriteHash(@sprites)
  end

  def pbEndScene
    pbDisposeSpriteHash(@sprites)
    @viewport.dispose
  end
end

module FallenTime 
  def self.halloween?
    return true if $beta_tester
    return false 
  end

  def self.halloween? 
    return true if $beta_tester
    return false 
    return true if (getRealTime.day == 31 && getRealTime.month == 10) || getRealTime.month == 11
    return true if getRealTime.month == 12 && getRealTime.day <= 23 
    return false 
  end

  def self.children_day
    return true if $DEBUG && $fallen || $beta_tester
    return if $offline 
    initial_time = Time.new(2024, 10, 12, 0, 0, 0)
    final_time   = Time.new(2024, 10, 22, 23, 59, 59)
    today = Time.new($real_day.year,$real_day.month, $real_day.day, $real_day.hour, $real_day.min, $real_day.sec)
    if (today >= initial_time) && (today <= final_time)
      return true 
    else
      return false 
    end
  end
end
